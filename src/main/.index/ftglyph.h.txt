
struct FT_Glyph { [x]
    FT_Library            library; [x] FTLibrary getLibrary()
    FT_Glyph_Format       format;  [x] FTGlyphFormat getFormat()
    FT_Vector             advance; [x] FTVector getAdvance()
}

struct FT_BitmapGlyph { [x]
    FT_GlyphRec root;   [x] FTGlyph getRoot()
    FT_Int      left;   [x] int getLeft()
    FT_Int      top;    [x] int getTop()
    FT_Bitmap   bitmap; [x] FTBitmap getBitmap()
}

struct FT_OutlineGlyph { [ ]
    FT_GlyphRec root;    [ ] FTGlyph getRoot()
    FT_Outline  outline; [ ] FTOutline getOutline()
}

struct FT_SvgGlyph { [ ]
    FT_GlyphRec     root;                [ ] FTGlyph getRoot()
    FT_Byte*        svg_document;        [ ] byte[] getSvgDocument()
    FT_ULong        svg_document_length; [ ] long getSvgDocumentLength()
    FT_UInt         glyph_index;         [ ] long getGlyphIndex()
    FT_Size_Metrics metrics;             [ ] FTSizeMetrics getMetrics()
    FT_UShort       units_per_EM;        [ ] int getUnitsPerEM()
    FT_UShort       start_glyph_id;      [ ] int getStartGlyphID()
    FT_UShort       end_glyph_id;        [ ] int getEndGlyphID()
    FT_Matrix       transform;           [ ] FTMatrix getTransform()
    FT_Vector       delta;               [ ] FTVector getDelta()
}

enum FT_Glyph_BBox_Mode { [x]
    UNSCALED  = 0
    SUBPIXELS = 0
    GRIDFIT   = 1
    TRUNCATE  = 2
    PIXELS    = 3
}
